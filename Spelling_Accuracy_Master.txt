export default function App() {
  // Navigation & Mode States
  const [activeTab, setActiveTab] = useState('practice'); // 'practice', 'quiz', 'custom', 'mistakes'
  const [selectedCategory, setSelectedCategory] = useState('All');

  // Quiz / Practice Gameplay State
  const [currentWord, setCurrentWord] = useState(null);
  const [userAnswer, setUserAnswer] = useState(null); // 'yes', 'no', or null
  const [feedback, setFeedback] = useState(null); // { isCorrect: bool, message: string }
  const [stats, setStats] = useState({
    totalReviewed: 0,
    correctCount: 0,
    streak: 0,
    bestStreak: 0,
    timedHighScore: 0
  });

  // Timed Quiz specific states
  const [quizActive, setQuizActive] = useState(false);
  const [timeLeft, setTimeLeft] = useState(60);
  const [quizScore, setQuizScore] = useState(0);

  // Custom Word List State
  const [customWords, setCustomWords] = useState([
    { word: 'Pharaoh', isCorrect: true, correctSpelling: 'Pharaoh', phonetic: '/ˈfeɪ.roʊ/', definition: 'A ruler in ancient Egypt.', example: 'Tutankhamun was an ancient Egyptian pharaoh.', category: 'Custom', note: 'Ends in -aoh.' },
    { word: 'Pharoah', isCorrect: false, correctSpelling: 'Pharaoh', phonetic: '/ˈfeɪ.roʊ/', definition: 'A ruler in ancient Egypt.', example: 'Tutankhamun was an ancient Egyptian pharaoh.', category: 'Custom', note: 'Misspelled -oah instead of -aoh.' }
  ]);
  const [newWordInput, setNewWordInput] = useState({ word: '', isCorrect: true, correctSpelling: '', definition: '', note: '' });

  // Mistakes Bank
  const [mistakes, setMistakes] = useState([]);

  // Audio Speech state
  const [isSpeaking, setIsSpeaking] = useState(false);

  // Categories list
  const categories = ['All', 'Commonly Confused', 'Tricky Double Letters', 'Academic/SAT', 'Silent Letters', 'Business', 'Custom'];

  // Initialize first word
  const getRandomWord = useCallback(() => {
    let pool = WORD_DATABASE;
    if (activeTab === 'custom') {
      pool = customWords.length > 0 ? customWords : WORD_DATABASE;
    } else if (selectedCategory !== 'All') {
      pool = WORD_DATABASE.filter(w => w.category === selectedCategory);
    }

    if (!pool || pool.length === 0) pool = WORD_DATABASE;
    const randomIndex = Math.floor(Math.random() * pool.length);
    return pool[randomIndex];
  }, [selectedCategory, activeTab, customWords]);

  useEffect(() => {
    setCurrentWord(getRandomWord());
  }, [selectedCategory, activeTab, getRandomWord]);

  // Handle Speech Synthesis
  const speakWord = (text) => {
    if (!('speechSynthesis' in window)) {
      alert("Text-to-speech is not supported in this browser.");
      return;
    }
    window.speechSynthesis.cancel(); // Stop ongoing speech
    const utterance = new SpeechSynthesisUtterance(text);
    utterance.rate = 0.85; // Slightly slower for clarity
    utterance.onstart = () => setIsSpeaking(true);
    utterance.onend = () => setIsSpeaking(false);
    utterance.onerror = () => setIsSpeaking(false);
    window.speechSynthesis.speak(utterance);
  };

  // Process user choice (Yes / No)
  const handleAnswer = (choice) => {
    if (!currentWord || feedback !== null) return; // Prevent double taps during feedback preview

    const choiceIsCorrect = (choice === 'yes' && currentWord.isCorrect) || (choice === 'no' && !currentWord.isCorrect);
    
    setUserAnswer(choice);
    
    // Update Stats
    setStats(prev => {
      const newTotal = prev.totalReviewed + 1;
      const newCorrect = choiceIsCorrect ? prev.correctCount + 1 : prev.correctCount;
      const newStreak = choiceIsCorrect ? prev.streak + 1 : 0;
      const newBestStreak = Math.max(newStreak, prev.bestStreak);
      return {
        ...prev,
        totalReviewed: newTotal,
        correctCount: newCorrect,
        streak: newStreak,
        bestStreak: newBestStreak
      };
    });

    if (quizActive) {
      if (choiceIsCorrect) setQuizScore(prev => prev + 100 + stats.streak * 10);
    }

    // Set feedback box
    setFeedback({
      isCorrect: choiceIsCorrect,
      message: choiceIsCorrect 
        ? "Spot on! That's correct." 
        : `Oops! "${currentWord.word}" is actually ${currentWord.isCorrect ? 'SPELLED CORRECTLY' : 'MISSPELLED'}.`
    });

    // Save to mistakes bank if wrong
    if (!choiceIsCorrect) {
      setMistakes(prev => {
        if (prev.some(m => m.word === currentWord.word)) return prev;
        return [currentWord, ...prev];
      });
    }

    // Speak correct pronunciation automatically on answer
    speakWord(currentWord.correctSpelling);
  };

  // Next Question Handler
  const handleNextWord = () => {
    setFeedback(null);
    setUserAnswer(null);
    setCurrentWord(getRandomWord());
  };

  // Keyboard Shortcuts Support
  useEffect(() => {
    const handleKeyDown = (e) => {
      // Avoid firing when typing in form inputs
      if (['INPUT', 'TEXTAREA'].includes(document.activeElement.tagName)) return;

      if (feedback === null) {
        if (e.key === 'ArrowLeft' || e.key === 'y' || e.key === 'Y') {
          handleAnswer('yes');
        } else if (e.key === 'ArrowRight' || e.key === 'n' || e.key === 'N') {
          handleAnswer('no');
        } else if (e.code === 'Space') {
          e.preventDefault();
          if (currentWord) speakWord(currentWord.correctSpelling);
        }
      } else {
        if (e.key === 'Enter' || e.key === 'ArrowRight' || e.key === ' ') {
          e.preventDefault();
          handleNextWord();
        }
      }
    };

    window.addEventListener('keydown', handleKeyDown);
    return () => window.removeEventListener('keydown', handleKeyDown);
  }, [feedback, currentWord, getRandomWord]);

  // Timer logic for Timed Challenge
  useEffect(() => {
    let timer = null;
    if (quizActive && timeLeft > 0) {
      timer = setInterval(() => setTimeLeft(prev => prev - 1), 1000);
    } else if (timeLeft === 0 && quizActive) {
      setQuizActive(false);
      setStats(prev => ({
        ...prev,
        timedHighScore: Math.max(prev.timedHighScore, quizScore)
      }));
    }
    return () => clearInterval(timer);
  }, [quizActive, timeLeft, quizScore]);

  // Start Speed Challenge
  const startTimedQuiz = () => {
    setQuizActive(true);
    setTimeLeft(60);
    setQuizScore(0);
    setFeedback(null);
    setUserAnswer(null);
    setCurrentWord(getRandomWord());
  };

  // Add Custom Word Handler
  const handleAddCustomWord = (e) => {
    e.preventDefault();
    if (!newWordInput.word.trim()) return;

    const formatted = {
      word: newWordInput.word.trim(),
      isCorrect: newWordInput.isCorrect,
      correctSpelling: newWordInput.correctSpelling.trim() || newWordInput.word.trim(),
      phonetic: '/custom/',
      definition: newWordInput.definition.trim() || 'Custom user added word.',
      example: 'User custom vocabulary list item.',
      category: 'Custom',
      note: newWordInput.note.trim() || 'Custom entry'
    };

    setCustomWords([formatted, ...customWords]);
    setNewWordInput({ word: '', isCorrect: true, correctSpelling: '', definition: '', note: '' });
  };

  // Calculate Accuracy
  const accuracyPercentage = stats.totalReviewed > 0 
    ? Math.round((stats.correctCount / stats.totalReviewed) * 100) 
    : 0;

  return (
    <div className="min-h-screen bg-slate-950 text-slate-100 flex flex-col font-sans selection:bg-indigo-500 selection:text-white">
      {/* Header Bar */}
      <header className="border-b border-slate-800 bg-slate-900/80 backdrop-blur-md sticky top-0 z-50">
        <div className="max-w-6xl mx-auto px-4 h-16 flex items-center justify-between">
          <div className="flex items-center space-x-3">
            <div className="bg-gradient-to-tr from-indigo-500 to-purple-600 p-2.5 rounded-xl shadow-lg shadow-indigo-500/20">
              <Sparkles className="w-5 h-5 text-white" />
            </div>
            <div>
              <h1 className="font-bold text-lg leading-tight bg-gradient-to-r from-white via-slate-200 to-slate-400 bg-clip-text text-transparent">
                SpellCheck AI
              </h1>
              <p className="text-xs text-slate-400">Interactive Spelling Trainer</p>
            </div>
          </div>

          {/* Navigation Tabs */}
          <nav className="hidden sm:flex items-center space-x-1 bg-slate-950/60 p-1 rounded-xl border border-slate-800">
            <button
              onClick={() => { setActiveTab('practice'); setQuizActive(false); }}
              className={`px-4 py-1.5 rounded-lg text-sm font-medium transition-all ${
                activeTab === 'practice' 
                  ? 'bg-indigo-600 text-white shadow-md' 
                  : 'text-slate-400 hover:text-slate-200 hover:bg-slate-800/50'
              }`}
            >
              Practice Mode
            </button>
            <button
              onClick={() => { setActiveTab('quiz'); setQuizActive(false); }}
              className={`px-4 py-1.5 rounded-lg text-sm font-medium transition-all flex items-center space-x-1.5 ${
                activeTab === 'quiz' 
                  ? 'bg-indigo-600 text-white shadow-md' 
                  : 'text-slate-400 hover:text-slate-200 hover:bg-slate-800/50'
              }`}
            >
              <Zap className="w-3.5 h-3.5 text-amber-400" />
              <span>Speed Challenge</span>
            </button>
            <button
              onClick={() => { setActiveTab('custom'); setQuizActive(false); }}
              className={`px-4 py-1.5 rounded-lg text-sm font-medium transition-all ${
                activeTab === 'custom' 
                  ? 'bg-indigo-600 text-white shadow-md' 
                  : 'text-slate-400 hover:text-slate-200 hover:bg-slate-800/50'
              }`}
            >
              Custom List
            </button>
            <button
              onClick={() => { setActiveTab('mistakes'); setQuizActive(false); }}
              className={`px-4 py-1.5 rounded-lg text-sm font-medium transition-all flex items-center space-x-1.5 ${
                activeTab === 'mistakes' 
                  ? 'bg-indigo-600 text-white shadow-md' 
                  : 'text-slate-400 hover:text-slate-200 hover:bg-slate-800/50'
              }`}
            >
              <span>Mistakes</span>
              {mistakes.length > 0 && (
                <span className="bg-rose-500/20 text-rose-400 border border-rose-500/30 text-xs px-1.5 rounded-full font-bold">
                  {mistakes.length}
                </span>
              )}
            </button>
          </nav>
        </div>
      </header>

      {/* Main Container */}
      <main className="flex-1 max-w-5xl w-full mx-auto p-4 sm:p-6 flex flex-col justify-between">
        
        {/* Top Mini Dashboard Stats */}
        <div className="grid grid-cols-2 sm:grid-cols-4 gap-3 mb-6">
          <div className="bg-slate-900/60 border border-slate-800 rounded-xl p-3 flex items-center space-x-3">
            <div className="p-2 bg-indigo-500/10 text-indigo-400 rounded-lg">
              <BarChart2 className="w-5 h-5" />
            </div>
            <div>
              <p className="text-xs text-slate-400 font-medium">Accuracy</p>
              <p className="text-lg font-bold text-slate-100">{accuracyPercentage}%</p>
            </div>
          </div>

          <div className="bg-slate-900/60 border border-slate-800 rounded-xl p-3 flex items-center space-x-3">
            <div className="p-2 bg-amber-500/10 text-amber-400 rounded-lg">
              <Flame className="w-5 h-5" />
            </div>
            <div>
              <p className="text-xs text-slate-400 font-medium">Current Streak</p>
              <p className="text-lg font-bold text-amber-400">{stats.streak} 🔥</p>
            </div>
          </div>

          <div className="bg-slate-900/60 border border-slate-800 rounded-xl p-3 flex items-center space-x-3">
            <div className="p-2 bg-emerald-500/10 text-emerald-400 rounded-lg">
              <Trophy className="w-5 h-5" />
            </div>
            <div>
              <p className="text-xs text-slate-400 font-medium">Best Streak</p>
              <p className="text-lg font-bold text-slate-100">{stats.bestStreak}</p>
            </div>
          </div>

          <div className="bg-slate-900/60 border border-slate-800 rounded-xl p-3 flex items-center space-x-3">
            <div className="p-2 bg-purple-500/10 text-purple-400 rounded-lg">
              <BookOpen className="w-5 h-5" />
            </div>
            <div>
              <p className="text-xs text-slate-400 font-medium">Total Reviewed</p>
              <p className="text-lg font-bold text-slate-100">{stats.totalReviewed}</p>
            </div>
          </div>
        </div>

        {/* TAB 1: PRACTICE MODE & MAIN CARD */}
        {(activeTab === 'practice' || activeTab === 'custom') && (
          <div className="flex-1 flex flex-col justify-center max-w-2xl mx-auto w-full">
            
            {/* Category selector pills for Practice */}
            {activeTab === 'practice' && (
              <div className="flex items-center space-x-2 overflow-x-auto pb-3 mb-2 no-scrollbar">
                <span className="text-xs font-semibold text-slate-400 uppercase tracking-wider whitespace-nowrap mr-1">Filter:</span>
                {categories.map((cat) => (
                  <button
                    key={cat}
                    onClick={() => setSelectedCategory(cat)}
                    className={`px-3 py-1 rounded-full text-xs font-medium whitespace-nowrap transition-all ${
                      selectedCategory === cat 
                        ? 'bg-slate-200 text-slate-900 font-semibold' 
                        : 'bg-slate-900 text-slate-400 border border-slate-800 hover:border-slate-700 hover:text-slate-200'
                    }`}
                  >
                    {cat}
                  </button>
                ))}
              </div>
            )}

            {/* Word Challenge Card */}
            {currentWord ? (
              <div className={`relative bg-slate-900/90 border rounded-3xl p-6 sm:p-8 shadow-2xl transition-all duration-300 ${
                feedback === null 
                  ? 'border-slate-800 shadow-indigo-950/20' 
                  : feedback.isCorrect 
                    ? 'border-emerald-500/50 bg-emerald-950/10 shadow-emerald-900/20' 
                    : 'border-rose-500/50 bg-rose-950/10 shadow-rose-900/20'
              }`}>
                
                {/* Top Badge & Audio trigger */}
                <div className="flex items-center justify-between mb-6">
                  <span className="px-3 py-1 rounded-full bg-slate-800 text-indigo-400 border border-slate-700/60 text-xs font-medium uppercase tracking-wider">
                    {currentWord.category || 'General'}
                  </span>

                  <button
                    onClick={() => speakWord(currentWord.correctSpelling)}
                    className={`flex items-center space-x-2 px-3 py-1.5 rounded-xl border text-xs font-medium transition-all ${
                      isSpeaking 
                        ? 'bg-indigo-600 text-white border-indigo-500 animate-pulse' 
                        : 'bg-slate-800/80 border-slate-700 text-slate-300 hover:bg-slate-700 hover:text-white'
                    }`}
                    title="Listen to Pronunciation (Shortcut: Space)"
                  >
                    <Volume2 className="w-4 h-4" />
                    <span>{isSpeaking ? 'Listening...' : 'Listen'}</span>
                  </button>
                </div>

                {/* Main Prominent Word Display */}
                <div className="text-center py-6">
                  <p className="text-xs uppercase tracking-widest text-slate-400 mb-2">Is this word spelled correctly?</p>
                  <h2 className="text-4xl sm:text-5xl font-extrabold tracking-tight text-white mb-2 font-serif">
                    {currentWord.word}
                  </h2>
                  <p className="text-sm font-mono text-slate-400">{currentWord.phonetic}</p>
                </div>

                {/* Action Buttons: YES / NO */}
                {feedback === null ? (
                  <div className="grid grid-cols-2 gap-4 mt-6">
                    <button
                      onClick={() => handleAnswer('yes')}
                      className="group relative flex flex-col items-center justify-center p-4 sm:p-5 rounded-2xl bg-emerald-600/10 hover:bg-emerald-600/20 border-2 border-emerald-500/40 hover:border-emerald-500 text-emerald-400 transition-all duration-200 shadow-lg shadow-emerald-950/30 active:scale-[0.98]"
                    >
                      <div className="flex items-center space-x-2 font-bold text-lg mb-1">
                        <Check className="w-6 h-6 stroke-[3]" />
                        <span>YES, Correct</span>
                      </div>
                      <span className="text-[10px] uppercase font-semibold tracking-wider text-emerald-500/80 bg-emerald-950/60 px-2 py-0.5 rounded border border-emerald-800/40">
                        Press Y or ←
                      </span>
                    </button>

                    <button
                      onClick={() => handleAnswer('no')}
                      className="group relative flex flex-col items-center justify-center p-4 sm:p-5 rounded-2xl bg-rose-600/10 hover:bg-rose-600/20 border-2 border-rose-500/40 hover:border-rose-500 text-rose-400 transition-all duration-200 shadow-lg shadow-rose-950/30 active:scale-[0.98]"
                    >
                      <div className="flex items-center space-x-2 font-bold text-lg mb-1">
                        <X className="w-6 h-6 stroke-[3]" />
                        <span>NO, Misspelled</span>
                      </div>
                      <span className="text-[10px] uppercase font-semibold tracking-wider text-rose-500/80 bg-rose-950/60 px-2 py-0.5 rounded border border-rose-800/40">
                        Press N or →
                      </span>
                    </button>
                  </div>
                ) : (
                  /* Answer Feedback Banner & Explanation */
                  <div className="mt-4 space-y-4 animate-in fade-in slide-in-from-bottom-2 duration-200">
                    <div className={`p-4 rounded-2xl border flex items-start space-x-3 ${
                      feedback.isCorrect 
                        ? 'bg-emerald-500/10 border-emerald-500/30 text-emerald-300' 
                        : 'bg-rose-500/10 border-rose-500/30 text-rose-300'
                    }`}>
                      {feedback.isCorrect ? (
                        <CheckCircle className="w-6 h-6 text-emerald-400 flex-shrink-0 mt-0.5" />
                      ) : (
                        <XCircle className="w-6 h-6 text-rose-400 flex-shrink-0 mt-0.5" />
                      )}
                      <div>
                        <h3 className="font-bold text-base mb-0.5">{feedback.message}</h3>
                        <p className="text-xs opacity-90">{currentWord.note}</p>
                      </div>
                    </div>

                    {/* Dictionary Card Info */}
                    <div className="bg-slate-950/60 rounded-2xl p-4 border border-slate-800 text-slate-300 text-sm space-y-2">
                      <div>
                        <span className="text-xs font-semibold uppercase tracking-wider text-slate-500 block">Correct Spelling</span>
                        <p className="text-lg font-bold text-indigo-300">{currentWord.correctSpelling}</p>
                      </div>
                      <div>
                        <span className="text-xs font-semibold uppercase tracking-wider text-slate-500 block">Definition</span>
                        <p className="text-slate-300">{currentWord.definition}</p>
                      </div>
                      {currentWord.example && (
                        <div>
                          <span className="text-xs font-semibold uppercase tracking-wider text-slate-500 block">Example Usage</span>
                          <p className="italic text-slate-400 font-serif">"{currentWord.example}"</p>
                        </div>
                      )}
                    </div>

                    {/* Next Word Button */}
                    <button
                      onClick={handleNextWord}
                      className="w-full py-3.5 bg-indigo-600 hover:bg-indigo-500 text-white font-semibold rounded-2xl shadow-lg shadow-indigo-600/30 flex items-center justify-center space-x-2 transition-all active:scale-[0.99]"
                    >
                      <span>Continue to Next Word</span>
                      <ArrowRight className="w-4 h-4" />
                    </button>
                  </div>
                )}

                {/* Keyboard tip footer */}
                <div className="mt-6 pt-4 border-t border-slate-800/80 flex items-center justify-between text-xs text-slate-500">
                  <span className="flex items-center space-x-1">
                    <HelpCircle className="w-3.5 h-3.5" />
                    <span>Shortcuts: Y / ← (Yes), N / → (No), Space (Audio)</span>
                  </span>
                  <span>Press Enter to skip</span>
                </div>
              </div>
            ) : (
              <div className="text-center py-12 bg-slate-900 border border-slate-800 rounded-3xl">
                <p className="text-slate-400 mb-4">No words available in this category.</p>
                <button 
                  onClick={() => setSelectedCategory('All')} 
                  className="px-4 py-2 bg-indigo-600 text-white text-sm rounded-xl"
                >
                  Reset Category Filter
                </button>
              </div>
            )}
          </div>
        )}

        {/* TAB 2: SPEED CHALLENGE / TIMED QUIZ */}
        {activeTab === 'quiz' && (
          <div className="max-w-2xl mx-auto w-full flex-1 flex flex-col justify-center">
            {!quizActive ? (
              <div className="bg-slate-900 border border-slate-800 rounded-3xl p-8 text-center space-y-6 shadow-2xl">
                <div className="inline-flex p-4 bg-amber-500/10 text-amber-400 rounded-2xl mb-2">
                  <Zap className="w-10 h-10" />
                </div>
                <div>
                  <h2 className="text-2xl font-bold text-white mb-2">60-Second Speed Blitz</h2>
                  <p className="text-slate-400 text-sm max-w-md mx-auto">
                    Test your spelling accuracy against the clock. Answer as many words correctly as possible before time runs out!
                  </p>
                </div>

                <div className="grid grid-cols-2 gap-4 max-w-xs mx-auto text-left bg-slate-950 p-4 rounded-2xl border border-slate-800">
                  <div>
                    <p className="text-xs text-slate-500">High Score</p>
                    <p className="text-xl font-bold text-amber-400">{stats.timedHighScore} pts</p>
                  </div>
                  <div>
                    <p className="text-xs text-slate-500">Time Limit</p>
                    <p className="text-xl font-bold text-slate-200">60 sec</p>
                  </div>
                </div>

                <button
                  onClick={startTimedQuiz}
                  className="w-full py-4 bg-gradient-to-r from-amber-500 to-amber-600 hover:from-amber-400 hover:to-amber-500 text-slate-950 font-bold text-lg rounded-2xl shadow-xl shadow-amber-500/20 transition-all flex items-center justify-center space-x-2"
                >
                  <Play className="w-5 h-5 fill-slate-950" />
                  <span>Start Challenge Now</span>
                </button>
              </div>
            ) : (
              <div className="space-y-4">
                {/* Timer & Score Header */}
                <div className="flex items-center justify-between bg-slate-900 border border-slate-800 p-4 rounded-2xl">
                  <div className="flex items-center space-x-2 text-amber-400">
                    <Clock className="w-5 h-5" />
                    <span className="font-mono text-2xl font-bold">{timeLeft}s</span>
                  </div>
                  <div className="text-right">
                    <span className="text-xs text-slate-500 uppercase tracking-wider block">Score</span>
                    <span className="text-2xl font-extrabold text-white">{quizScore}</span>
                  </div>
                </div>

                {/* Main Challenge Card Embedded in Quiz */}
                {currentWord && (
                  <div className="bg-slate-900 border border-slate-800 rounded-3xl p-6 text-center space-y-6">
                    <div className="py-4">
                      <p className="text-xs uppercase tracking-widest text-slate-500 mb-1">Is this correctly spelled?</p>
                      <h2 className="text-4xl font-extrabold text-white font-serif">{currentWord.word}</h2>
                    </div>

                    <div className="grid grid-cols-2 gap-4">
                      <button
                        onClick={() => { handleAnswer('yes'); setTimeout(handleNextWord, 250); }}
                        className="py-4 rounded-2xl bg-emerald-600/20 hover:bg-emerald-600/30 border border-emerald-500/40 text-emerald-400 font-bold text-lg"
                      >
                        YES
                      </button>
                      <button
                        onClick={() => { handleAnswer('no'); setTimeout(handleNextWord, 250); }}
                        className="py-4 rounded-2xl bg-rose-600/20 hover:bg-rose-600/30 border border-rose-500/40 text-rose-400 font-bold text-lg"
                      >
                        NO
                      </button>
                    </div>
                  </div>
                )}
              </div>
            )}
          </div>
        )}

        {/* TAB 3: CUSTOM WORD LIST MANAGER */}
        {activeTab === 'custom' && (
          <div className="max-w-3xl mx-auto w-full space-y-6">
            {/* Form to add custom word */}
            <div className="bg-slate-900 border border-slate-800 p-6 rounded-3xl space-y-4">
              <h3 className="font-bold text-lg text-white flex items-center space-x-2">
                <Plus className="w-5 h-5 text-indigo-400" />
                <span>Add Custom Word to Practice</span>
              </h3>

              <form onSubmit={handleAddCustomWord} className="space-y-4">
                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
                  <div>
                    <label className="block text-xs text-slate-400 mb-1">Target Word / Spelling to Test</label>
                    <input
                      type="text"
                      required
                      placeholder="e.g., Acommodate"
                      value={newWordInput.word}
                      onChange={e => setNewWordInput({...newWordInput, word: e.target.value})}
                      className="w-full bg-slate-950 border border-slate-800 rounded-xl px-3.5 py-2 text-sm text-white focus:outline-none focus:border-indigo-500"
                    />
                  </div>

                  <div>
                    <label className="block text-xs text-slate-400 mb-1">Is this version Correct or Misspelled?</label>
                    <select
                      value={newWordInput.isCorrect ? 'true' : 'false'}
                      onChange={e => setNewWordInput({...newWordInput, isCorrect: e.target.value === 'true'})}
                      className="w-full bg-slate-950 border border-slate-800 rounded-xl px-3.5 py-2 text-sm text-white focus:outline-none focus:border-indigo-500"
                    >
                      <option value="true">Correctly Spelled Word</option>
                      <option value="false">Misspelled / Deliberate Error</option>
                    </select>
                  </div>
                </div>

                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
                  <div>
                    <label className="block text-xs text-slate-400 mb-1">Actual Correct Spelling</label>
                    <input
                      type="text"
                      placeholder="e.g., Accommodate"
                      value={newWordInput.correctSpelling}
                      onChange={e => setNewWordInput({...newWordInput, correctSpelling: e.target.value})}
                      className="w-full bg-slate-950 border border-slate-800 rounded-xl px-3.5 py-2 text-sm text-white focus:outline-none focus:border-indigo-500"
                    />
                  </div>

                  <div>
                    <label className="block text-xs text-slate-400 mb-1">Definition or Memory Hint</label>
                    <input
                      type="text"
                      placeholder="e.g., Double C and Double M"
                      value={newWordInput.note}
                      onChange={e => setNewWordInput({...newWordInput, note: e.target.value})}
                      className="w-full bg-slate-950 border border-slate-800 rounded-xl px-3.5 py-2 text-sm text-white focus:outline-none focus:border-indigo-500"
                    />
                  </div>
                </div>

                <button
                  type="submit"
                  className="w-full py-2.5 bg-indigo-600 hover:bg-indigo-500 text-white font-medium rounded-xl text-sm transition-all"
                >
                  Add to Custom Vocabulary List
                </button>
              </form>
            </div>

            {/* Existing custom list preview */}
            <div className="bg-slate-900 border border-slate-800 rounded-3xl p-6">
              <div className="flex items-center justify-between mb-4">
                <h4 className="font-bold text-white text-sm">Your Custom List ({customWords.length})</h4>
                {customWords.length > 0 && (
                  <button
                    onClick={() => setActiveTab('practice')}
                    className="text-xs text-indigo-400 hover:underline flex items-center space-x-1"
                  >
                    <span>Practice Custom Words</span>
                    <ArrowRight className="w-3 h-3" />
                  </button>
                )}
              </div>

              {customWords.length === 0 ? (
                <p className="text-xs text-slate-500 text-center py-6">No custom words added yet.</p>
              ) : (
                <div className="divide-y divide-slate-800">
                  {customWords.map((cw, idx) => (
                    <div key={idx} className="py-3 flex items-center justify-between">
                      <div>
                        <div className="flex items-center space-x-2">
                          <span className="font-semibold text-white">{cw.word}</span>
                          <span className={`text-[10px] px-2 py-0.5 rounded-full border ${
                            cw.isCorrect 
                              ? 'bg-emerald-500/10 border-emerald-500/30 text-emerald-400' 
                              : 'bg-rose-500/10 border-rose-500/30 text-rose-400'
                          }`}>
                            {cw.isCorrect ? 'Valid' : 'Misspelled Test'}
                          </span>
                        </div>
                        <p className="text-xs text-slate-400 mt-0.5">Correct: {cw.correctSpelling} • {cw.note}</p>
                      </div>

                      <button
                        onClick={() => setCustomWords(customWords.filter((_, i) => i !== idx))}
                        className="text-slate-500 hover:text-rose-400 p-1.5"
                        title="Delete word"
                      >
                        <Trash2 className="w-4 h-4" />
                      </button>
                    </div>
                  ))}
                </div>
              )}
            </div>
          </div>
        )}

        {/* TAB 4: MISTAKES REVIEW BANK */}
        {activeTab === 'mistakes' && (
          <div className="max-w-2xl mx-auto w-full space-y-4">
            <div className="flex items-center justify-between mb-2">
              <div>
                <h2 className="text-xl font-bold text-white">Mistakes Review Bank</h2>
                <p className="text-xs text-slate-400">Revisit words you struggled with during this session</p>
              </div>

              {mistakes.length > 0 && (
                <button
                  onClick={() => setMistakes([])}
                  className="text-xs text-slate-400 hover:text-rose-400 flex items-center space-x-1 bg-slate-900 border border-slate-800 px-3 py-1.5 rounded-xl"
                >
                  <RefreshCw className="w-3.5 h-3.5" />
                  <span>Clear History</span>
                </button>
              )}
            </div>

            {mistakes.length === 0 ? (
              <div className="text-center py-12 bg-slate-900 border border-slate-800 rounded-3xl p-8">
                <Award className="w-12 h-12 text-indigo-400 mx-auto mb-3 opacity-60" />
                <h3 className="font-bold text-white mb-1">No Mistakes Recorded!</h3>
                <p className="text-xs text-slate-400">Either you haven't played yet or you have a 100% perfect score!</p>
              </div>
            ) : (
              <div className="space-y-3">
                {mistakes.map((m, idx) => (
                  <div key={idx} className="bg-slate-900 border border-slate-800 rounded-2xl p-4 flex flex-col sm:flex-row sm:items-center justify-between gap-3">
                    <div className="space-y-1">
                      <div className="flex items-center space-x-2">
                        <span className="font-serif font-bold text-lg text-white">{m.word}</span>
                        <span className="text-xs text-slate-400 font-mono">({m.phonetic})</span>
                      </div>
                      <p className="text-xs text-indigo-300">Correct: <span className="font-semibold">{m.correctSpelling}</span></p>
                      <p className="text-xs text-slate-400">{m.definition}</p>
                    </div>

                    <button
                      onClick={() => speakWord(m.correctSpelling)}
                      className="self-start sm:self-center px-3 py-1.5 bg-slate-800 hover:bg-slate-700 text-slate-200 border border-slate-700 text-xs font-medium rounded-xl flex items-center space-x-1.5 flex-shrink-0"
                    >
                      <Volume2 className="w-3.5 h-3.5" />
                      <span>Listen</span>
                    </button>
                  </div>
                ))}
              </div>
            )}
          </div>
        )}

      </main>

      {/* Footer / Shortcut indicator */}
      <footer className="border-t border-slate-900 py-3 text-center text-xs text-slate-600">
        SpellCheck AI Interactive Learning • React & Tailwind CSS
      </footer>
    </div>
  );
}
